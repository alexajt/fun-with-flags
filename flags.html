viewof game_mode = Inputs.radio(
  ["Normal (Random Countries)", "Twin Trap (Lookalike Flags!)"],
  { label: "Difficulty:", value: "Twin Trap (Lookalike Flags!)" }
)
mutable score = 0
mutable streak = 0
mutable round = 1
mutable answered = null
mutable target = flag_list[Math.floor(Math.random() * flag_list.length)]
// Build 4 answer choices for the current round
choices = {
  round;
  const current = target;
  let distractors;
  if (game_mode.startsWith("Twin")) {
    distractors = [
      { country: current.sim1_name, file: current.sim1_file },
      { country: current.sim2_name, file: current.sim2_file },
      { country: current.sim3_name, file: current.sim3_file }
    ];
  } else {
    const others = flag_list.filter(d => d.country !== current.country);
    distractors = others
      .sort(() => Math.random() - 0.5)
      .slice(0, 3)
      .map(d => ({ country: d.country, file: d.file }));
  }
  return [
    { country: current.country, file: current.file, correct: true },
    ...distractors.map(d => ({ ...d, correct: false }))
  ].sort(() => Math.random() - 0.5);
}
{
  const container = html`
    <div style="background: #f8f9fa; color: #212529; padding: 1.5rem; border-radius: 12px; border: 1px solid #dee2e6; text-align: center; margin-top: 0.75rem;">
      <div style="display: flex; justify-content: space-between; margin-bottom: 1rem; font-weight: 600; font-size: 1.05rem; color: #212529;">
        <span>Round: ${round}</span>
        <span>Score: ${score}</span>
        <span>🔥 Streak: ${streak}</span>
      </div>

      <img src="w2560/${target.file}" style="max-width: 360px; width: 100%; height: 210px; object-fit: contain; border: 1px solid #ccc; border-radius: 6px; background: white; padding: 0.5rem; box-shadow: 0 3px 8px rgba(0,0,0,0.08);">

      <p style="margin: 1rem 0 0.75rem; font-weight: 600; color: #212529;">Which country or territory does this flag belong to?</p>

      <div id="btn-grid" style="display: grid; grid-template-columns: repeat(2, 1fr); gap: 0.75rem; max-width: 520px; margin: 0 auto;"></div>

      <div id="feedback" style="margin-top: 1rem; min-height: 2.5rem;"></div>
    </div>
  `;

  const grid = container.querySelector("#btn-grid");
  const feedback = container.querySelector("#feedback");

  choices.forEach(choice => {
    const btn = document.createElement("button");
    btn.type = "button";
    btn.textContent = choice.country;
    btn.style.cssText = "padding: 0.75rem 1rem; border-radius: 8px; border: 2px solid #adb5bd; background-color: #ffffff; color: #212529 !important; font-weight: 600; cursor: pointer; font-size: 1rem; line-height: 1.3;";

    if (answered !== null) {
      btn.disabled = true;
      btn.style.cursor = "default";
      if (choice.correct) {
        btn.style.backgroundColor = "#d1e7dd";
        btn.style.borderColor = "#198754";
        btn.style.color = "#0f5132";
      } else if (answered === choice.country) {
        btn.style.backgroundColor = "#f8d7da";
        btn.style.borderColor = "#dc3545";
        btn.style.color = "#842029";
      }
    } else {
      btn.onclick = () => {
        mutable answered = choice.country;
        if (choice.correct) {
          mutable score = score + 1;
          mutable streak = streak + 1;
        } else {
          mutable streak = 0;
        }
      };
    }
    grid.appendChild(btn);
  });

  if (answered !== null) {
    const isCorrect = answered === target.country;
    const msg = html`<div style="margin-bottom: 0.75rem; font-weight: 600; color: ${isCorrect ? '#198754' : '#dc3545'};">
      ${isCorrect ? "✅ Spot on!" : `❌ Not quite — that flag is ${target.country}!`}
    </div>`;
    const nextBtn = html`<button type="button" style="padding: 0.55rem 1.25rem; border-radius: 8px; border: none; background-color: #0d6efd; color: #ffffff; font-weight: 600; cursor: pointer;">Next Flag →</button>`;
    nextBtn.onclick = () => {
      mutable answered = null;
      mutable target = flag_list[Math.floor(Math.random() * flag_list.length)];
      mutable round = round + 1;
    };
    feedback.appendChild(msg);
    feedback.appendChild(nextBtn);
  }

  return container;
}
